def fibonacci(n: int) -> int:
    """返回第 n 个斐波那契数，f(0)=0, f(1)=1。"""
    if n < 0:
        raise ValueError("n 不能为负数")
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return a

if __name__ == "__main__":
    print(fibonacci(10))  # 55
```
